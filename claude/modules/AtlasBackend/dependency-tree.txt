main.py
├── config.settings
├── database.Base, engine
├── routers.sources

│   ├── models.Source, Domain, Project, WebPage

│   ├── models.Domain, Source, Project, WebPage

├── routers.projects
│   ├── models.Project, Domain, WebPage
│   └── schemas.Project*
├── routers.pages
│   ├── models.WebPage, Project, Chunk
│   ├── schemas.WebPage*
│   └── services.indexer.index_web_page
│         ├── services.embeddings.embed_texts
│         │     └── openai (+ settings)
│         └── models.Chunk (+ hierarchy join for denormalized ids)
└── routers.search

    └── POST search → services.rag.generate_answer
          ├── services.embeddings.embed_texts / cosine helpers
          ├── models.Chunk (+ page/hierarchy for citations)
          └── openai chat (+ settings)
